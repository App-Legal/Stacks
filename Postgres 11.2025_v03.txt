  # ============================================================
  # ==== BLOCO: BACKUP POSTGRES ====
  # Backup lógico diário às 03:00.
  # ============================================================
  postgres-backup:
    image: postgres:16-alpine
    hostname: postgres-backup

    entrypoint: /bin/sh
    command:
      - -c
      - |
        set -e
        apk add --no-cache bash dcron findutils tar gzip > /dev/null 2>&1
        mkdir -p /data/bkp

        cat > /usr/local/bin/postgres-backup.sh << 'EOFPG'
        #!/bin/bash
        set -e
        BACKUP_DIR="/data/bkp"
        DATE=$$(date +%Y%m%d)
        TIMESTAMP=$$(date +%Y%m%d_%H%M%S)
        RETENTION_DAYS=7

        mkdir -p "$${BACKUP_DIR}"

        echo "[$$(date '+%Y-%m-%d %H:%M:%S')] Iniciando backup PostgreSQL..."

        if PGPASSWORD="$${POSTGRES_PASSWORD}" pg_isready -h "$${POSTGRES_HOST}" -U "$${POSTGRES_USER}" -d "$${POSTGRES_DB}" > /dev/null 2>&1; then
          PGPASSWORD="$${POSTGRES_PASSWORD}" pg_dump \
            -h "$${POSTGRES_HOST}" \
            -U "$${POSTGRES_USER}" \
            -d "$${POSTGRES_DB}" \
            -F c -b -v \
            -f "$${BACKUP_DIR}/postgres_$${TIMESTAMP}.dump" 2>&1 | grep -v "^pg_dump:" || true

          if [ -f "$${BACKUP_DIR}/postgres_$${TIMESTAMP}.dump" ]; then
            tar -czf "$${BACKUP_DIR}/postgres_backup_$${DATE}.tar.gz" -C "$${BACKUP_DIR}" "postgres_$${TIMESTAMP}.dump"
            rm -f "$${BACKUP_DIR}/postgres_$${TIMESTAMP}.dump"
          else
            echo "ERRO: arquivo de dump PostgreSQL não gerado"
          fi
        else
          echo "ERRO: conexão com PostgreSQL falhou"
        fi

        find "$${BACKUP_DIR}" -name "postgres_backup_*.tar.gz" -type f -mtime +$${RETENTION_DAYS} -delete 2>/dev/null || true

        echo "[$$(date '+%Y-%m-%d %H:%M:%S')] Backup PostgreSQL finalizado."
        EOFPG

        chmod +x /usr/local/bin/postgres-backup.sh
        echo "0 3 * * * /usr/local/bin/postgres-backup.sh >> /var/log/postgres-backup.log 2>&1" > /etc/crontabs/root

        /usr/local/bin/postgres-backup.sh || true
        crond -f -l 2
