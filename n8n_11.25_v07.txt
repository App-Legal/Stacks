  ####
  # 💾 POSTGRES BACKUP - Backup Automático Diário
  ####
  postgres-backup:
    image: postgres:15.7-alpine
    hostname: postgres-backup
    environment:
      POSTGRES_HOST: postgres
      POSTGRES_DB: n8ndb
      POSTGRES_USER: n8n
      POSTGRES_PASSWORD: P@ssw0rd_pg_2025
      BACKUP_KEEP_DAYS: 7
      TZ: America/Sao_Paulo
    volumes:
      - postgres_backup:/backups
    networks:
      - netvps
    entrypoint: /bin/sh
    command:
      - -c
      - |
        set -e
        apk add --no-cache dcron postgresql-client gzip
        cat > /usr/local/bin/backup.sh << 'EOF'
        #!/bin/sh
        set -e
        TIMESTAMP=$$(date +%Y%m%d_%H%M%S)
        BACKUP_FILE="/backups/n8ndb_$${TIMESTAMP}.sql.gz"
        echo "🚀 Iniciando backup: $${BACKUP_FILE}"
        PGPASSWORD="$${POSTGRES_PASSWORD}" pg_dump -h "$${POSTGRES_HOST}" -U "$${POSTGRES_USER}" -d "$${POSTGRES_DB}" --format=plain --no-owner --no-acl 2>&1 | gzip > "$${BACKUP_FILE}"
        [ -f "$${BACKUP_FILE}" ] && echo "✅ Backup concluído: $$(du -h "$${BACKUP_FILE}" | cut -f1)" || exit 1
        find /backups -name "n8ndb_*.sql.gz" -type f -mtime +$${BACKUP_KEEP_DAYS} -delete
        echo "📦 Backups disponíveis:"
        ls -lh /backups/n8ndb_*.sql.gz 2>/dev/null || echo "Nenhum backup encontrado"
        EOF
        chmod +x /usr/local/bin/backup.sh
        echo "0 2 * * * /usr/local/bin/backup.sh >> /var/log/backup.log 2>&1" > /etc/crontabs/root
        echo "✅ Executando backup inicial..."
        /usr/local/bin/backup.sh
        echo "⏰ Agendamento configurado: diariamente às 02:00"
        crond -f -l 2
    deploy:
      mode: replicated
      replicas: 1
      placement:
        constraints:
          - node.role == manager
      restart_policy:
        condition: on-failure
      resources:
        limits:
          cpus: "0.5"
          memory: 256M
