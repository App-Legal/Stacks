  # BLOCO 6: BACKUP AUTOMÁTICO - PostgreSQL
  ################################################################################
  # ┌─────────────────────────────────────────────────────────────────────────┐
  # │ 💾 BACKUP AUTOMÁTICO                                                     │
  # ├─────────────────────────────────────────────────────────────────────────┤
  # │ FUNÇÃO:                                                                  │
  # │   - Backup automático do PostgreSQL a cada 6 horas                      │
  # │   - Compressão GZIP para economia de espaço                             │
  # │   - Retenção: últimos 7 backups (rotação automática)                    │
  # │                                                                          │
  # │ LOCALIZAÇÃO:                                                             │
  # │   - Host: /root/backups/evolution/                                      │
  # │   - Formato: evolution_backup_YYYYMMDD_HHMMSS.sql.gz                    │
  # │                                                                          │
  # │ SCRIPT INLINE:                                                           │
  # │   - Loop infinito com sleep de 6 horas (21600s)                         │
  # │   - pg_dump com compressão                                              │
  # │   - Limpeza automática de backups antigos                               │
  # │   - Logs de sucesso/erro                                                │
  # │                                                                          │
  # │ RECURSOS:                                                                │
  # │   - Mínimos: 0.1 CPU / 64MB RAM                                         │
  # │   - Não impacta performance da aplicação                                │
  # └─────────────────────────────────────────────────────────────────────────┘
  
  backup:
    image: postgres:16-alpine
    
    networks:
      - evo_network
    
    volumes:
      - postgres_data:/var/lib/postgresql/data:ro
      - /root/backups/evolution:/backups
    
    environment:
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: "Pg#2024!Evo@Secure"
      POSTGRES_DB: postgres
      PGHOST: evo_1125_v3_postgres
      PGPORT: "5432"
    
    entrypoint: |
      sh -c '
      echo "🔄 Backup automático iniciado..."
      
      mkdir -p /backups
      
      while true; do
        TIMESTAMP=$$(date +%Y%m%d_%H%M%S)
        BACKUP_FILE="/backups/evolution_backup_$$TIMESTAMP.sql.gz"
        
        echo "📦 Criando backup: $$BACKUP_FILE"
        
        PGPASSWORD="Pg#2024!Evo@Secure" pg_dump \
          -h evo_1125_v3_postgres \
          -U postgres \
          -d postgres \
          --no-owner \
          --no-acl \
          | gzip > "$$BACKUP_FILE"
        
        if [ $$? -eq 0 ]; then
          echo "✅ Backup criado com sucesso!"
          ls -t /backups/evolution_backup_*.sql.gz | tail -n +8 | xargs -r rm
          echo "🧹 Backups antigos removidos (mantendo últimos 7)"
        else
          echo "❌ Erro ao criar backup"
        fi
        
        echo "⏰ Próximo backup em 6 horas..."
        sleep 21600
      done
      '
    
    deploy:
      mode: replicated
      replicas: 1
      placement:
        constraints:
          - node.role == manager
      
      resources:
        limits:
          cpus: '0.5'
          memory: 256M
        reservations:
          cpus: '0.1'
          memory: 64M

        condition: on-failure
